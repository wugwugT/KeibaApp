### 実装内容

#### Toast.tsx（新規）

```typescript
import { useEffect } from 'react';
import { Animated, StyleSheet, Text } from 'react-native';

type Props = {
  message: string;
  visible: boolean;
  onHide: () => void;
  duration?: number;
};

export const Toast = ({ message, visible, onHide, duration = 2000 }: Props) => {
  const opacity = new Animated.Value(0);

  useEffect(() => {
    if (visible) {
      Animated.sequence([
        Animated.timing(opacity, { toValue: 1, duration: 200, useNativeDriver: true }),
        Animated.delay(duration),
        Animated.timing(opacity, { toValue: 0, duration: 200, useNativeDriver: true }),
      ]).start(() => onHide());
    }
  }, [visible]);

  if (!visible) return null;

  return (
    <Animated.View style={[styles.container, { opacity }]}>
      <Text style={styles.text}>{message}</Text>
    </Animated.View>
  );
};

const styles = StyleSheet.create({
  container: {
    position: 'absolute',
    bottom: 100,
    alignSelf: 'center',
    backgroundColor: 'rgba(0,0,0,0.8)',
    paddingHorizontal: 20,
    paddingVertical: 12,
    borderRadius: 24,
  },
  text: { color: '#fff', fontSize: 14 },
});
```

### 実装内容

#### useMonthlyStats.ts（新規）

```typescript
type MonthlyRow = {
  month: string;        // "2026-01" 形式
  investment: number;
  return: number;
  profit: number;
  recoveryRate: number;
  betCount: number;     // 賭け回数
};

export const useMonthlyStats = (): MonthlyRow[] => {
  const [stats, setStats] = useState<MonthlyRow[]>([]);

  useFocusEffect(useCallback(() => {
    const load = async () => {
      const records = await getAllBetRecords();
      const map = new Map<string, { inv: number; ret: number; count: number }>();

      records.forEach((r) => {
        const month = r.date.toISOString().slice(0, 7); // "YYYY-MM"
        const cur = map.get(month) ?? { inv: 0, ret: 0, count: 0 };
        cur.inv += r.investment;
        cur.ret += r.return;
        cur.count += 1;
        map.set(month, cur);
      });

      const rows: MonthlyRow[] = [...map.entries()].map(([month, v]) => ({
        month,
        investment: v.inv,
        return: v.ret,
        profit: v.ret - v.inv,
        recoveryRate: v.inv > 0 ? Math.round((v.ret / v.inv) * 100) : 0,
        betCount: v.count,
      }));

      setStats(rows);
    };
    load();
  }, []));

  return stats;
};
```

import * as FileSystem from 'expo-file-system';
import * as Sharing from 'expo-sharing';

export const exportAsCSV = async (records: BetRecord[]): Promise<void> => {
  const header = 'date,place,race_no,bet_type,investment,return\n';
  const rows = records.map(r =>
    `${r.date.toISOString()},${r.place},${r.race_no},${r.bet_type},${r.investment},${r.return}`
  ).join('\n');

  const filePath = `${FileSystem.cacheDirectory}keiba_records.csv`;
  await FileSystem.writeAsStringAsync(filePath, header + rows);
  await Sharing.shareAsync(filePath);
};

export const exportAsJSON = async (records: BetRecord[]): Promise<void> => {
  const json = JSON.stringify(records, null, 2);
  const filePath = `${FileSystem.cacheDirectory}keiba_records.json`;
  await FileSystem.writeAsStringAsync(filePath, json);
  await Sharing.shareAsync(filePath);
};

import { Component, ReactNode } from 'react';
import { View, Text, TouchableOpacity, StyleSheet } from 'react-native';

type Props = { children: ReactNode };
type State = { hasError: boolean; error: Error | null };

export class ErrorBoundary extends Component<Props, State> {
  state: State = { hasError: false, error: null };

  static getDerivedStateFromError(error: Error) {
    return { hasError: true, error };
  }

  handleReset = () => {
    this.setState({ hasError: false, error: null });
  };

  render() {
    if (this.state.hasError) {
      return (
        <View style={styles.container}>
          <Text style={styles.title}>エラーが発生しました</Text>
          <Text style={styles.message}>
            アプリで予期しないエラーが発生しました。
          </Text>
          <TouchableOpacity style={styles.button} onPress={this.handleReset}>
            <Text style={styles.buttonText}>再試行</Text>
          </TouchableOpacity>
        </View>
      );
    }
    return this.props.children;
  }
}

const styles = StyleSheet.create({
  container: { flex: 1, justifyContent: 'center', alignItems: 'center', padding: 24 },
  title: { fontSize: 20, fontWeight: '600', marginBottom: 8 },
  message: { fontSize: 14, opacity: 0.7, textAlign: 'center', marginBottom: 24 },

## #15 QR読み取り時の日付反映（P0）⚠️ 暫定対応

### 調査結果
- QRコードのbase部分に含まれるのは開催スケジュール情報:
  - `year`（7-8桁目）: 西暦下2桁（例: 26 → 2026年）
  - `round`（9-10桁目）: 第N回開催（JRA開催回数であり、月ではない）
  - `day`（11-12桁目）: 開催N日目（開催日数であり、カレンダーの日ではない）
- **QRコードにカレンダー日付は含まれない**ことが実データ（サンプル4: 1回1日目 = 1月4日）で確認済み
- base42桁の未解析領域（4-6桁: `000`、23-28桁: ランダム値）にも日付情報なし

### 現状の実装（暫定対応）
- QRからの日付自動設定は行わない（誤った日付を設定するより安全）
- RecordEdit画面に `QR情報: 2026年 1回 1日目 — 正しい日付を設定してください` とヒント表示
- 日付欄の枠をオレンジ色にして手動設定が必要なことを視覚的に示す

### 将来対応: JRA開催カレンダーマスタ

```
開催カレンダーJSON:
{
  "2026": {
    "中山": [
      { "kai": 1, "days": ["2026-01-04", "2026-01-05", "2026-01-11", ...] },
      { "kai": 2, "days": ["2026-02-28", "2026-03-01", ...] },
    ],
    "東京": [...],
    ...
  }
}

→ year=26, place="中山", kai=1, day=1 → "2026-01-04"（1回中山1日目）
```

JRAの年間スケジュールをマスタデータとして持ち、`year` + `place` + `round` + `day` から
正確な開催日を逆引きする。毎年更新が必要。
